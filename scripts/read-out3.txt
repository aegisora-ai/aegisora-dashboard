export default async function BlogPost({ params }: Props) {
  const { locale, slug } = await params;
  const post = POSTS.find((p) => p.slug === slug);
  if (!post) notFound();
  const t = await getTranslations({ locale, namespace: "Blog" });
  const fmt = (iso: string) =>
    new Date(iso + "T00:00:00Z").toLocaleDateString(locale === "de" ? "de-DE" : "en-GB", {
      day: "numeric",
      month: "long",
      year: "numeric",
      timeZone: "UTC",
    });

  const more = POSTS.filter((p) => p.slug !== post.slug).slice(0, 3);

  return (
    <>
      <PageHero
        eyebrow={post.category + " \u00b7 " + fmt(post.date) + " \u00b7 " + t("minRead", { n: post.readMinutes })}
        title={post.title}
        intro={post.excerpt}
      />
      <article className="mx-auto max-w-3xl px-4 py-14 sm:px-6 lg:py-20">
        {locale !== "en" ? <p className="mb-6 text-sm text-muted-foreground">{t("enOnly")}</p> : null}
        {post.body.map((b, i) => {
          if (b.t === "h2") return <h2 key={i} className="mt-10 text-2xl font-medium tracking-[-0.01em] text-foreground">{b.text}</h2>;
          if (b.t === "p") return <p key={i} className="mt-4 text-base leading-relaxed text-foreground/75 sm:text-lg">{b.text}</p>;
          if (b.t === "ul")
            return (
              <ul key={i} className="mt-4 list-disc space-y-2 pl-6 text-base leading-relaxed text-foreground/75 sm:text-lg marker:text-primary">
                {b.items.map((it) => <li key={it}>{it}</li>)}
              </ul>
            );
          return (
            <pre key={i} className="mt-4 overflow-x-auto rounded-xl border border-border bg-surface-1 p-4 font-mono text-sm text-foreground">{b.text}</pre>
          );
        })}
      </article>
      <Section title={t("more")} tone="alt">
        <CardGrid
          cols={3}
          items={more.map((p) => ({
            n: p.category,
            title: p.title,
            body: p.excerpt,
            href: "/blog/" + p.slug,
          }))}
        />
      </Section>
      <CtaBand
        title={t("postCtaTitle")}
        body={t("postCtaBody")}
        primary={{ label: t("test"), href: "/signup" }}
        secondary={{ label: t("enterprise"), href: "/enterprise" }}
      />
    </>
  );
}
