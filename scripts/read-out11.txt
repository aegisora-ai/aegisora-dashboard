type Counts = { agents: number | null; policies: number | null; pending: number | null; incidents: number | null; total: number | null; block: number | null; escalate: number | null };
type Bucket = { allow: number; escalate: number; block: number };
type Point = { created_at: string; decision: string | null };

const HOUR = 36e5;
const num = (v: number | null) => (v === null ? "-" : v.toLocaleString("en-US"));
const cnt = (r: { count: number | null; error: unknown }) => (r.error ? null : (r.count ?? 0));

function bucketize(points: Point[]): { buckets: Bucket[]; labels: string[] } {
  const now = Date.now();
  const buckets: Bucket[] = Array.from({ length: 24 }, () => ({ allow: 0, escalate: 0, block: 0 }));
  const labels = Array.from({ length: 24 }, (_, i) => String(new Date(now - (23 - i) * HOUR).getHours()).padStart(2, "0"));
  for (const p of points) {
    const idx = 23 - Math.floor((now - new Date(p.created_at).getTime()) / HOUR);
    if (idx < 0 || idx > 23) continue;
    if (p.decision === "ALLOW") buckets[idx].allow++;
    else if (p.decision === "ESCALATE") buckets[idx].escalate++;
    else if (p.decision === "BLOCK") buckets[idx].block++;
  }
  return { buckets, labels };
}

function timeLeft(iso: string) {
  const mins = Math.round((Date.parse(iso) - Date.now()) / 6e4);
  if (mins < 1) return "expires now";
  if (mins < 60) return mins + "m left";
  return Math.round(mins / 60) + "h left";
}

function Bars({ buckets, labels }: { buckets: Bucket[]; labels: string[] }) {
  const totals = buckets.map((b) => b.allow + b.escalate + b.block);
  const max = Math.max(1, ...totals);
  return (
    <>
      <div className="aegc-bars" role="img" aria-label="Decisions per hour, last 24 hours">
        {buckets.map((b, i) => {
          const t = totals[i];
          const h = t === 0 ? 1 : Math.max(6, (t / max) * 100);
          return (
            <div key={i} className="aegc-bar" title={labels[i] + ":00 \u00b7 " + t + (t === 1 ? " decision" : " decisions")}>
              <div className="aegc-bar__area">
                <div className="aegc-bar__col" style={{ height: h + "%", animationDelay: i * 20 + "ms" }}>
                  {t > 0 ? (
                    <>
                      <i style={{ flex: b.block, background: "var(--block)" }} />
                      <i style={{ flex: b.escalate, background: "var(--escalate)" }} />
                      <i style={{ flex: b.allow, background: "var(--allow)" }} />
                    </>
                  ) : null}
                </div>
              </div>
              <span>{i % 3 === 0 ? labels[i] : ""}</span>
            </div>
          );
        })}
      </div>
      <div className="aegc-legend">
        <span><i style={{ background: "var(--allow)" }} />Allow</span>
        <span><i style={{ background: "var(--escalate)" }} />Escalate</span>
        <span><i style={{ background: "var(--block)" }} />Block</span>
      </div>
    </>
  );
}

function Overview({ ws }: { ws: Workspace }) {

  const [agents, setAgents] = useState<Row[]>([]);
  const [pend, setPend] = useState<Row[]>([]);
  const [hide, setHide] = useState(false);

  useEffect(() => {
    let alive = true;
    const load = async () => {
      const sb = createClient();
      const now = new Date();
      const nowIso = now.toISOString();
      const since = new Date(now.getTime() - 24 * HOUR).toISOString();
      const w = ws.id;
      const [ag, po, pe, inc, tot, blk, esc, ser, rec, agl, pel] = await Promise.all([
        sb.from("agents").select("id", { count: "exact", head: true }).eq("workspace_id", w),
        sb.from("policies").select("id", { count: "exact", head: true }).eq("workspace_id", w),
        sb.from("approvals").select("id", { count: "exact", head: true }).eq("workspace_id", w).eq("status", "pending").gt("expires_at", nowIso),
        sb.from("incidents").select("id", { count: "exact", head: true }).eq("workspace_id", w).eq("status", "open"),

"use client";

import { useState } from "react";
import { Check, Pencil, Plus, Trash2 } from "lucide-react";
import { Link } from "@/i18n/navigation";
import { createClient } from "@/lib/supabase/client";
import { useWorkspace } from "@/components/aegisora/workspace-provider";

const BTN = "inline-flex h-8 items-center gap-1.5 rounded-full border border-border px-3 text-xs hover:bg-surface-3 disabled:opacity-50";
const RED = { color: "var(--block)", borderColor: "rgb(255 92 92 / 0.4)" } as const;

export default function WorkspacePage() {
  const { workspaces, currentWorkspace, loading, setCurrentWorkspaceId, refresh } = useWorkspace();
  const [editId, setEditId] = useState<string | null>(null);
  const [name, setName] = useState("");
  const [delId, setDelId] = useState<string | null>(null);
  const [confirm, setConfirm] = useState("");
  const [busy, setBusy] = useState(false);
  const [msg, setMsg] = useState<{ ok: boolean; text: string } | null>(null);

  const isOwner = (r: string) => r.toUpperCase() === "OWNER";
  const canEdit = (r: string) => isOwner(r) || r.toUpperCase() === "ADMIN";

  async function rename(id: string) {
    const v = name.trim();
    if (v.length < 2) { setMsg({ ok: false, text: "Name must be at least 2 characters." }); return; }
    setBusy(true);
    setMsg(null);
    const { data, error } = await createClient().from("workspaces").update({ name: v }).eq("id", id).select("id");
    setBusy(false);
    if (error || !data || data.length === 0) {
      setMsg({ ok: false, text: error?.message ?? "Rename was not permitted for this workspace." });
      return;
    }
    setEditId(null);
    setMsg({ ok: true, text: "Workspace renamed." });
    refresh();
  }

  async function remove(id: string) {
    setBusy(true);
    setMsg(null);
    const { data, error } = await createClient().from("workspaces").delete().eq("id", id).select("id");
    setBusy(false);
    if (error || !data || data.length === 0) {
      setMsg({ ok: false, text: error?.message ?? "Delete was not permitted. Only owners can delete a workspace." });
      return;
    }
    setDelId(null);
    setConfirm("");
    setMsg({ ok: true, text: "Workspace deleted." });
    refresh();
  }

  if (loading) {
    return <div className="mx-auto max-w-4xl py-12 text-sm text-muted-foreground">Loading workspaces...</div>;
  }

  return (
    <div className="mx-auto max-w-4xl space-y-6">
      <header className="flex flex-wrap items-center justify-between gap-3">
        <div>
          <h1 className="text-3xl font-light tracking-tight">Workspaces</h1>
          <p className="mt-1 text-sm text-muted-foreground">Switch, rename, delete or create workspaces.</p>
        </div>
        <Link href="/onboarding" className="aegc-btn aegc-btn--solid"><Plus aria-hidden="true" />New workspace</Link>
      </header>

      {msg ? (
        <p role="status" className="rounded-lg border border-border px-4 py-3 text-sm" style={{ color: msg.ok ? "var(--allow)" : "var(--block)" }}>{msg.text}</p>
      ) : null}

      {workspaces.length === 0 ? (
        <p className="aeg-panel p-5 text-sm text-muted-foreground">No workspace yet. Create one to get started.</p>
      ) : (
        <ul className="aeg-panel divide-y divide-border/60">
          {workspaces.map((w) => {
            const current = currentWorkspace?.id === w.id;
            return (
              <li key={w.id} className="p-4">
                <div className="flex flex-wrap items-center gap-3">
                  <div className="min-w-0 flex-1">
                    {editId === w.id ? (
                      <input
                        value={name}
                        onChange={(e) => setName(e.target.value)}
                        maxLength={60}
                        autoFocus
                        aria-label="Workspace name"
                        className="h-9 w-full max-w-xs rounded-md border border-border bg-transparent px-3 text-sm"
                      />
                    ) : (
                      <p className="truncate text-sm font-medium">{w.name}{current ? <Check className="ml-2 inline size-4 text-cyan" aria-label="Current" /> : null}</p>
                    )}
                    <p className="mt-0.5 truncate font-mono text-[11px] text-muted-foreground">{w.slug} &middot; {w.role}</p>
                  </div>
                  <div className="flex flex-wrap items-center gap-2">
                    {editId === w.id ? (
                      <>
                        <button type="button" className={BTN} disabled={busy} onClick={() => rename(w.id)}>Save</button>
                        <button type="button" className={BTN} disabled={busy} onClick={() => setEditId(null)}>Cancel</button>
                      </>
                    ) : (
                      <>
                        {!current ? <button type="button" className={BTN} onClick={() => setCurrentWorkspaceId(w.id)}>Switch</button> : null}
                        {canEdit(w.role) ? (
                          <button type="button" className={BTN} onClick={() => { setEditId(w.id); setName(w.name); setDelId(null); setMsg(null); }}>
                            <Pencil className="size-3.5" aria-hidden="true" />Rename
                          </button>
                        ) : null}
                        {isOwner(w.role) ? (
                          <button type="button" className={BTN} style={RED} onClick={() => { setDelId(w.id); setConfirm(""); setEditId(null); setMsg(null); }}>
                            <Trash2 className="size-3.5" aria-hidden="true" />Delete
                          </button>
                        ) : null}
                      </>
                    )}
                  </div>
                </div>

                {delId === w.id ? (
                  <div className="mt-4 rounded-lg border p-4" style={{ borderColor: "rgb(255 92 92 / 0.35)" }}>
                    <p className="text-sm">
                      This permanently deletes <b>{w.name}</b> with all its agents, policies, decisions, evidence and keys. This cannot be undone.
                    </p>
                    <p className="mt-2 text-xs text-muted-foreground">Type DELETE to confirm.</p>
                    <div className="mt-2 flex flex-wrap items-center gap-2">
                      <input
                        value={confirm}
                        onChange={(e) => setConfirm(e.target.value)}
                        aria-label="Type DELETE to confirm"
                        className="h-9 w-40 rounded-md border border-border bg-transparent px-3 text-sm"
                      />
                      <button type="button" className={BTN} style={RED} disabled={busy || confirm !== "DELETE"} onClick={() => remove(w.id)}>Delete workspace</button>
                      <button type="button" className={BTN} disabled={busy} onClick={() => { setDelId(null); setConfirm(""); }}>Cancel</button>
                    </div>
                  </div>
                ) : null}
              </li>
            );
          })}
        </ul>
      )}

      <p className="text-sm text-muted-foreground">
        Members and roles are managed under <Link className="underline underline-offset-4" href="/dashboard/team">Team</Link>.
      </p>
    </div>
  );
}

"use client";

import { useState, type FormEvent } from "react";
import Link from "next/link";
import { createClient } from "@/lib/supabase/client";
import { LiveTable } from "@/components/aegisora/live/live-table";
import { Badge, Notice, Panel, btnGhost, fmt } from "@/components/aegisora/live/ui";
import type { Workspace } from "@/lib/aegisora/use-workspace";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { btnPrimary } from "@/lib/ui";
import { confirmDialog, notify } from "@/components/aegisora/live/feedback";

function KeyTools({ ws, reload }: { ws: Workspace; reload: () => void }) {
  const [name, setName] = useState("");
  const [fresh, setFresh] = useState<string | null>(null);
  const [err, setErr] = useState<string | null>(null);
  const [copied, setCopied] = useState(false);
  const [needsMfa, setNeedsMfa] = useState(false);

  async function create(e: FormEvent) {
    e.preventDefault();
    setErr(null); setCopied(false); setNeedsMfa(false);
    const { data, error } = await createClient().rpc("create_api_key", { ws: ws.id, key_name: name.trim() });
    if (error) {
      if (/forbidden/i.test(error.message)) {
        const { data: lvl } = await createClient().auth.mfa.getAuthenticatorAssuranceLevel();
        const mfaMissing = lvl?.currentLevel !== "aal2";
        setNeedsMfa(mfaMissing);
        return setErr(mfaMissing ? "Multi-factor verification is required before you can manage API keys." : "Only workspace owners and admins can create API keys.");
      }
      return setErr(error.message);
    }
    setFresh(String(data)); setName(""); reload();
  }

  return (
    <div className="space-y-4">
      <Panel>
        <form onSubmit={create} className="flex flex-wrap items-end gap-3">
          <div className="min-w-56 flex-1 space-y-1.5"><Label htmlFor="k-name">Key name</Label><Input id="k-name" value={name} onChange={(e) => setName(e.target.value)} placeholder="production-runtime" required maxLength={80} /></div>
          <button type="submit" className={btnPrimary}>Create API key</button>
        </form>
      </Panel>
      {err && (
        <Notice>
          {err}
          {needsMfa && <>{" "}<Link href="/dashboard/identity" className="underline underline-offset-2">Verify on the Identity page</Link>.</>}
        </Notice>
      )}
      {fresh && (
        <Panel>
          <p className="text-sm font-medium">Copy your key now. It is shown only once and cannot be recovered.</p>
          <div className="mt-3 flex items-center gap-2">
            <code className="flex-1 overflow-x-auto rounded-sm bg-surface-3 px-3 py-2 font-mono text-xs">{fresh}</code>
            <button className={btnGhost} onClick={async () => { await navigator.clipboard.writeText(fresh); setCopied(true); }}>{copied ? "Copied" : "Copy"}</button>
          </div>
          <p className="mt-3 text-xs text-muted-foreground">Keep it server-side. Never ship it in browser code or commit it to git.</p>
        </Panel>
      )}
    </div>
  );
}

function stat(rows: Row[]) {
  let allow = 0, block = 0, esc = 0, policy = 0, high = 0;
  for (const r of rows) {
    if (r.decision === "ALLOW") allow++;
    else if (r.decision === "BLOCK") block++;
    else if (r.decision === "ESCALATE") esc++;

function build(rows: Row[]) {
  const days = Array.from({ length: 14 }, (_, i) => {
    const d = new Date();
    d.setHours(12, 0, 0, 0);
    d.setDate(d.getDate() - (13 - i));

  });

  const cur: Row[] = [];
  const prev: Row[] = [];
  const week = [0, 0, 0, 0, 0, 0, 0];
  for (const r of rows) {
    const t = new Date(r.created_at);
    const i = idx.get(dayKey(t));
    if (i === undefined) continue;
    const d = days[i];
    d.total++;
    if (r.decision === "ALLOW") d.allow++;
    else if (r.decision === "BLOCK") d.block++;

export function HourlyBars({ buckets, labels }: { buckets: Bucket[]; labels: string[] }) {
  const totals = buckets.map((b) => b.allow + b.escalate + b.block);
  const max = Math.max(1, ...totals);
  const peak = totals.some((t) => t > 0) ? totals.indexOf(Math.max(...totals)) : null;
  const [hover, setHover] = useState<number | null>(null);
  const active = hover ?? peak;
  const heightOf = (t: number) => (t === 0 ? 1.5 : Math.max(5, (t / max) * 100));
  const b = active !== null ? buckets[active] : null;

  return (
    <div>
      <div className="relative h-56" onPointerLeave={() => setHover(null)}>
        <div role="img" aria-label="Decisions per hour, last 24 hours" className="flex h-full items-end gap-1.5 sm:gap-2">

            <div key={i} className="flex h-full flex-1 cursor-pointer items-end" onPointerEnter={() => setHover(i)} onClick={() => setHover(i)}>
              <span
                className={cn("block w-full rounded-t-md transition-colors", i === active ? "bg-[#22d3ee] shadow-[0_0_28px_rgb(34_211_238/0.5)]" : "bg-[#1a6b78]")}

              />
            </div>
          ))}

          <div
            className="pointer-events-none absolute z-10 w-40 -translate-x-1/2 rounded-xl border border-border bg-popover p-3 text-xs shadow-xl"
            style={{
              left: "clamp(5rem, " + ((active + 0.5) / 24) * 100 + "%, calc(100% - 5rem))",

type R = { created_at: string; decision: string | null; action: string | null; risk_score: number | null };

const DAY = 864e5;
const C = { allow: "var(--allow)", escalate: "var(--escalate)", block: "var(--block)", primary: "var(--primary)", grid: "var(--border)" };

  contentStyle: { background: "var(--popover)", border: "1px solid var(--border)", borderRadius: 10, fontSize: 12, color: "var(--popover-foreground)", boxShadow: "0 8px 24px rgba(0,0,0,0.18)" },

const dayKey = (d: Date) => d.getFullYear() + "-" + String(d.getMonth() + 1).padStart(2, "0") + "-" + String(d.getDate()).padStart(2, "0");

function Card({ title, hint, className, children }: { title: string; hint?: string; className?: string; children: React.ReactNode }) {
  return (
    <section className={cn("aeg-panel min-w-0 p-4 sm:p-5", className)}>
      <h2 className="text-sm font-semibold">{title}</h2>

function build(rows: R[]) {
  const days = Array.from({ length: 14 }, (_, i) => {
    const d = new Date();
    d.setHours(12, 0, 0, 0);
    d.setDate(d.getDate() - (13 - i));
    return { key: dayKey(d), label: d.toLocaleDateString(undefined, { month: "short", day: "numeric" }), allow: 0, escalate: 0, block: 0 };
  });
  const idx = new Map(days.map((d, i) => [d.key, i]));
  const acts = new Map<string, { name: string; allow: number; escalate: number; block: number; total: number }>();
  const risk = [
    { name: "0-19", value: 0 }, { name: "20-39", value: 0 }, { name: "40-59", value: 0 },
    { name: "60-79", value: 0 }, { name: "80-100", value: 0 },
  ];
  for (const r of rows) {
    const i = idx.get(dayKey(new Date(r.created_at)));
    const k = r.decision === "ALLOW" ? "allow" : r.decision === "BLOCK" ? "block" : r.decision === "ESCALATE" ? "escalate" : null;
    if (i !== undefined && k) days[i][k]++;
    const name = r.action || "unknown";
    const a = acts.get(name) ?? { name, allow: 0, escalate: 0, block: 0, total: 0 };
    a.total++;
    if (k) a[k]++;
    acts.set(name, a);
    if (typeof r.risk_score === "number") risk[Math.min(4, Math.floor(r.risk_score / 20))].value++;
  }
  const top = [...acts.values()].sort((a, b) => b.total - a.total).slice(0, 6);
  return { days, top, risk, hasRisk: risk.some((x) => x.value > 0) };
}

          .eq("workspace_id", wsId)
          .gte("created_at", since)
          .order("created_at", { ascending: false })

        if (!alive) return;
        setRows(error ? [] : ((data ?? []) as R[]));
      } catch {
        if (alive) setRows([]);
      }
    })();
    return () => { alive = false; };
  }, [wsId]);

  const m = useMemo(() => (rows && rows.length > 0 ? build(rows) : null), [rows]);

  if (rows === null) return <div className="h-64 animate-pulse rounded-xl bg-surface-3" aria-busy="true" />;
  if (!m) return null;

  const topH = Math.max(200, m.top.length * 46 + 40);
