import { getTranslations } from "next-intl/server";
import { AlertTriangle, Info, Lightbulb } from "lucide-react";
import { slugify, type Block } from "@/lib/support-content";

export async function ArticleBody({ blocks, locale }: { blocks: Block[]; locale: string }) {
  const t = await getTranslations({ locale, namespace: "HelpCenter" });
  const CALL = {
    note: { Icon: Info, label: t("note"), cls: "border-border bg-surface-3" },
    tip: { Icon: Lightbulb, label: t("tip"), cls: "border-border bg-surface-3" },
    warn: { Icon: AlertTriangle, label: t("warn"), cls: "border-[#f0b429]/50 bg-[#f0b429]/10" },
  } as const;

  return (
    <div className="space-y-5">
      {blocks.map((b, i) => {
        if (b.t === "h")
          return (
            <h2 key={i} id={slugify(b.text)} className="aeg-display scroll-mt-24 pt-6 text-2xl sm:text-3xl">
              {b.text}
            </h2>
          );
        if (b.t === "p") return <p key={i} className="aeg-body-text text-base leading-7">{b.text}</p>;
        if (b.t === "ul")
          return (
            <ul key={i} className="aeg-body-text list-disc space-y-2 pl-6 text-base leading-7">
              {b.items.map((x) => <li key={x}>{x}</li>)}
            </ul>
          );
        if (b.t === "ol")
          return (
            <ol key={i} className="aeg-body-text list-decimal space-y-2 pl-6 text-base leading-7">
              {b.items.map((x) => <li key={x}>{x}</li>)}
            </ol>
          );
        const k = CALL[b.t];
        const Icon = k.Icon;
        return (
          <div key={i} className={"flex gap-3 rounded-xl border p-4 text-sm leading-6 " + k.cls}>
            <Icon className="mt-0.5 size-5 shrink-0" aria-hidden />
            <p><span className="font-semibold">{k.label}: </span>{b.text}</p>
          </div>
        );
      })}
    </div>
  );
}

"use client";

import { useState } from "react";
import { useTranslations } from "next-intl";
import { Check, Copy } from "lucide-react";

export function CopyPage({ text }: { text: string }) {
  const t = useTranslations("HelpCenter");
  const [done, setDone] = useState(false);
  async function copy() {
    try {
      await navigator.clipboard.writeText(text);
      setDone(true);
      window.setTimeout(() => setDone(false), 2000);
    } catch {
      setDone(false);
    }
  }
  return (
    <button
      type="button"
      onClick={copy}
      className="inline-flex h-9 items-center gap-2 rounded-lg border border-border px-3 text-sm font-semibold transition hover:bg-surface-3"
    >
      {done ? <Check className="size-4" aria-hidden /> : <Copy className="size-4" aria-hidden />}
      {done ? t("copied") : t("copy")}
    </button>
  );
}

"use client";

import { useEffect, useMemo, useRef, useState } from "react";
import { useTranslations } from "next-intl";
import { FileText, Search } from "lucide-react";

type Item = { title: string; summary: string; collection: string; href: string };

export function SupportSearch({ items }: { items: Item[] }) {
  const t = useTranslations("HelpCenter");
  const [q, setQ] = useState("");
  const [active, setActive] = useState(0);
  const ref = useRef<HTMLInputElement>(null);

  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === "k") {
        e.preventDefault();
        ref.current?.focus();
      }
    };
    document.addEventListener("keydown", onKey);
    return () => document.removeEventListener("keydown", onKey);
  }, []);

  const results = useMemo(() => {
    const term = q.trim().toLowerCase();
    if (term.length < 2) return [];
    return items
      .filter((i) => (i.title + " " + i.summary + " " + i.collection).toLowerCase().includes(term))
      .sort((a, b) => Number(!a.title.toLowerCase().includes(term)) - Number(!b.title.toLowerCase().includes(term)))
      .slice(0, 6);
  }, [q, items]);

  const onKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
    if (e.key === "ArrowDown") {
      e.preventDefault();
      setActive((i) => Math.min(i + 1, Math.max(results.length - 1, 0)));
    } else if (e.key === "ArrowUp") {
      e.preventDefault();
      setActive((i) => Math.max(i - 1, 0));
    } else if (e.key === "Enter" && results[active]) {
      window.location.assign(results[active].href);
    } else if (e.key === "Escape") {
      setQ("");
    }
  };

  return (
    <div className="relative max-w-xl">
      <div className="flex h-14 items-center gap-3 rounded-2xl border border-border bg-surface-1 px-4 shadow-sm focus-within:ring-2 focus-within:ring-primary/40">
        <Search className="size-5 shrink-0 text-muted-foreground" aria-hidden />
        <input
          ref={ref}
          value={q}
          onChange={(e) => {
            setQ(e.target.value);
            setActive(0);
          }}
          onKeyDown={onKeyDown}
          placeholder={t("searchPh")}
          aria-label={t("searchAria")}
          className="h-full min-w-0 flex-1 bg-transparent text-base outline-none placeholder:text-muted-foreground"
        />
        <kbd className="hidden rounded border border-border px-1.5 py-0.5 font-mono text-[11px] text-muted-foreground sm:inline">Ctrl K</kbd>
      </div>
      {q.trim().length >= 2 ? (
        <div className="absolute left-0 right-0 top-full z-20 mt-2 overflow-hidden rounded-2xl border border-border bg-surface-1 shadow-lg">
          {results.length === 0 ? (
            <p className="px-5 py-4 text-sm text-muted-foreground">{t("noResults")}</p>
          ) : (
            <ul>
              {results.map((r, i) => (
                <li key={r.href} className="border-b border-border last:border-b-0">
                  <a
                    href={r.href}
                    onMouseEnter={() => setActive(i)}
                    className={"flex items-start gap-3 px-5 py-3 transition " + (i === active ? "bg-surface-3" : "")}
                  >
                    <FileText className="mt-0.5 size-4 shrink-0 text-muted-foreground" aria-hidden />
                    <span>
                      <span className="block text-sm font-semibold">{r.title}</span>
                      <span className="mt-0.5 block text-xs text-muted-foreground">{r.collection} · {r.summary}</span>
                    </span>
                  </a>
                </li>
              ))}
            </ul>
          )}
        </div>
      ) : null}
    </div>
  );
}

type Item = { icon?: string; title: string; body?: string; href?: string; n?: string; [k: string]: unknown };

export function CardGrid({ cols = 3, items }: { cols?: number; items: Item[] }) {
  const n = (cols === 2 || cols === 4 || cols === 6 ? cols : 3) as 2 | 3 | 4 | 6;
  const perRow = n === 6 ? 3 : n;
  const missing = items.length % perRow === 0 ? 0 : perRow - (items.length % perRow);
  const icons = Lucide as unknown as Record<string, LucideIcon | undefined>;
  return (
    <Cells n={n}>
      {items.map((it, i) => {
        const Icon = it.icon ? icons[it.icon] : undefined;
        const inner = (
          <>
            {Icon ? <span className="aegx-ico"><Icon aria-hidden="true" /></span> : null}
            {it.n ? <span className="aegx-num">{it.n}</span> : null}
            <h3 className="aegx-h3">{it.title}</h3>
            {it.body ? <p className="aegx-p">{it.body}</p> : null}
          </>
        );
        return it.href ? (
          <Link key={i} href={it.href} className="aegx-cell">{inner}</Link>
        ) : (
          <div key={i} className="aegx-cell">{inner}</div>
        );
      })}
      {Array.from({ length: missing }).map((_, i) => (
        <div key={"f" + i} className="aegx-cell aegx-cell--fill" aria-hidden="true" />
      ))}
    </Cells>
  );
}
