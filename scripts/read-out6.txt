"use client";

import { useState } from "react";
import { useTranslations } from "next-intl";
import { Link, useRouter } from "@/i18n/navigation";
import { createClient } from "@/lib/supabase/client";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { btnPrimary } from "@/lib/ui";

export default function ResetPasswordPage() {
  const t = useTranslations("Auth");
  const router = useRouter();
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [done, setDone] = useState(false);

  async function onSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    setError(null);
    setLoading(true);
    const form = new FormData(e.currentTarget);
    const password = String(form.get("password") ?? "");

    const supabase = createClient();
    const { error: err } = await supabase.auth.updateUser({ password });
    setLoading(false);
    if (err) {
      setError(err.message);
      return;
    }
    setDone(true);
    setTimeout(() => router.push("/dashboard"), 1500);
  }

  return (
    <div className="w-full max-w-sm">
      <h1 className="bw-auth__h1">{t("reset.title")}</h1>
      <p className="mt-3 text-sm text-muted-foreground">{t("reset.desc")}</p>

      {error && (
        <p role="alert" className="mt-6 rounded-sm border border-block/30 bg-block-muted px-3 py-2 text-xs text-block">
          {error}
        </p>
      )}
      {done && (
        <p role="status" className="mt-6 rounded-sm border border-allow/30 bg-allow-muted px-3 py-2 text-xs text-allow">
          {t("reset.done")}
        </p>
      )}

      <form className="mt-6 space-y-4" onSubmit={onSubmit}>
        <div className="space-y-1.5">
          <Label htmlFor="password">{t("reset.label")}</Label>
          <Input id="password" name="password" type="password" autoComplete="new-password" required minLength={8} />
        </div>
        <button type="submit" disabled={loading || done} className={`${btnPrimary} w-full disabled:opacity-60`}>
          {loading ? t("reset.updating") : t("reset.submit")}
        </button>
      </form>

      <p className="mt-6 text-sm text-muted-foreground">
        <Link href="/login" className="text-primary underline-offset-4 hover:underline">{t("common.backToLogin")}</Link>
      </p>
    </div>
  );
}

type Pt = { x: number; y: number };
type Verdict = "allow" | "escalate" | "block";
type Packet = { from: number; verdict: Verdict; t: number; speed: number; hit: boolean };
type Burst = { x: number; y: number; age: number };

const AGENTS = 7;
const TAU = Math.PI * 2;
const RGB = {
  pre: "96,165,250",
  allow: "56,189,248",
  escalate: "240,180,41",
  block: "239,68,68",
  blue: "22,119,255",
  text: "169,187,219",
};

function quad(a: Pt, c: Pt, b: Pt, t: number): Pt {
  const u = 1 - t;
  return {
    x: u * u * a.x + 2 * u * t * c.x + t * t * b.x,
    y: u * u * a.y + 2 * u * t * c.y + t * t * b.y,
  };
}

function layout(w: number, h: number) {
  const wide = w > 980;
  const left = wide ? w * 0.42 : w * 0.06;
  const span = w * 0.94 - left;
  const cy = h * 0.5;
  const R = Math.max(26, Math.min(h * 0.14, span * 0.1));
  const gate: Pt = { x: left + span * 0.55, y: cy };
  const agents: Pt[] = [];
  const c1: Pt[] = [];
  const gateIn: Pt[] = [];
  for (let i = 0; i < AGENTS; i++) {
    const k = i / (AGENTS - 1);
    const a: Pt = { x: left + span * 0.05 * Math.sin(k * Math.PI), y: h * (0.15 + 0.7 * k) };
    agents.push(a);
    c1.push({ x: a.x + (gate.x - R - a.x) * 0.62, y: a.y });
    gateIn.push({ x: gate.x - R * 1.05, y: gate.y + (a.y - gate.y) * 0.1 });
  }
  const outs = {
    allow: { x: left + span * 0.97, y: cy - h * 0.2 } as Pt,
    escalate: { x: left + span * 0.97, y: cy + h * 0.2 } as Pt,
  };
  return { gate, R, agents, c1, gateIn, outs };
}
type Scene = ReturnType<typeof layout>;

function pos(p: Packet, s: Scene, t: number): Pt {
  if (t < 1 || p.verdict === "block") {
    const k = Math.max(0, Math.min(1, t));
    return quad(s.agents[p.from], s.c1[p.from], s.gateIn[p.from], k);
  }
  const o = p.verdict === "allow" ? s.outs.allow : s.outs.escalate;
  const a: Pt = { x: s.gate.x + s.R * 1.05, y: s.gate.y };
  const c: Pt = { x: a.x + (o.x - a.x) * 0.55, y: a.y };
  return quad(a, c, o, Math.min(1, t - 1));
}

"use client";

import { useState } from "react";
import { useTranslations } from "next-intl";
import { Link } from "@/i18n/navigation";
import { createClient } from "@/lib/supabase/client";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { btnPrimary } from "@/lib/ui";

type Field = { name: string; label: string; type: string; autoComplete?: string };
type Mode = "login" | "signup" | "reset" | "invite";

type Props = {
  mode: Mode;
  title: string;
  description: string;
  fields: Field[];
  submitLabel: string;
  redirectTo: string;
  links?: { text: string; label: string; href: string }[];
};

function withCompany(mode: Mode, fields: Field[], label: string): Field[] {
  if (mode !== "signup" || fields.some((f) => f.name === "company")) return fields;
  const companyField: Field = { name: "company", label, type: "text", autoComplete: "organization" };
  const i = fields.findIndex((f) => f.name === "password");
  return i === -1 ? [...fields, companyField] : [...fields.slice(0, i), companyField, ...fields.slice(i)];
}

export function AuthForm({ mode, title, description, fields, submitLabel, redirectTo, links }: Props) {
  const t = useTranslations("Auth");
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [notice, setNotice] = useState<string | null>(null);
  const allFields = withCompany(mode, fields, t("common.company"));

  async function onSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    setError(null);
    setNotice(null);
    setLoading(true);

    const form = new FormData(e.currentTarget);
    const email = String(form.get("email") ?? "").trim();
    const password = String(form.get("password") ?? "");
    const fullName = String(form.get("name") ?? "").trim();
    const company = String(form.get("company") ?? "").trim();

    const supabase = createClient();

    try {
      if (mode === "login") {
        const { error: err } = await supabase.auth.signInWithPassword({ email, password });
        if (err) throw err;
        window.location.href = redirectTo;
        return;
      }

      if (mode === "signup") {
        const { data, error: err } = await supabase.auth.signUp({
          email,
          password,
          options: { data: { full_name: fullName, ...(company ? { company } : {}) } },
        });
        if (err) throw err;
        if (data.session) {
          window.location.href = redirectTo;
        } else {
          setNotice(t("signup.checkInbox"));
        }
        return;
      }

      if (mode === "reset") {
        const { error: err } = await supabase.auth.resetPasswordForEmail(email, {
          redirectTo: `${window.location.origin}/reset-password`,
        });
        if (err) throw err;
        setNotice(t("forgot.sent"));
        return;
      }

      if (mode === "invite") {
        const { error: err } = await supabase.auth.updateUser({
          password,
          data: { full_name: fullName },
        });
        if (err) throw err;
        window.location.href = redirectTo;
        return;
      }
    } catch (err) {
      setError(err instanceof Error ? err.message : t("common.genericError"));
    } finally {
      setLoading(false);
    }
  }

  return (
    <div className="w-full max-w-sm">
      <h1 className="bw-auth__h1">{title}</h1>
      <p className="mt-3 text-sm text-muted-foreground">{description}</p>

      {error && (
        <p role="alert" className="mt-6 rounded-sm border border-block/30 bg-block-muted px-3 py-2 text-xs text-block">
          {error}
        </p>
      )}
      {notice && (

          {notice}
        </p>
      )}

        {allFields.map((f) => (

            <Label htmlFor={f.name}>{f.label}</Label>
            <Input id={f.name} name={f.name} type={f.type} autoComplete={f.autoComplete} required />
          </div>
        ))}
        <button type="submit" disabled={loading} className={`${btnPrimary} w-full disabled:opacity-60`}>
          {loading ? t("common.wait") : submitLabel}
        </button>
      </form>

      {links && (
        <ul className="mt-6 space-y-2 text-sm text-muted-foreground">
          {links.map((l) => (
            <li key={l.href}>
              {l.text}{" "}

                {l.label}
              </Link>
            </li>
          ))}
        </ul>
      )}
    </div>
  );
}

"use client";

import { useEffect, useRef, useState } from "react";
import { Eye, EyeOff } from "lucide-react";
import { useTranslations } from "next-intl";
import { Link } from "@/i18n/navigation";
import { createClient } from "@/lib/supabase/client";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { btnPrimary } from "@/lib/ui";
import { SocialButtons } from "./social-buttons";

const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

export function LoginForm({ redirectTo }: { redirectTo: string }) {
  const t = useTranslations("Auth");
  const [step, setStep] = useState<"email" | "password">("email");
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [show, setShow] = useState(false);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const mailRef = useRef<HTMLInputElement>(null);
  const pwRef = useRef<HTMLInputElement>(null);

  useEffect(() => {
    if (step === "password") pwRef.current?.focus();
    else mailRef.current?.focus();
  }, [step]);

  async function onSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    setError(null);

    if (step === "email") {
      if (!EMAIL_RE.test(email.trim())) {
        setError(t("login.invalidEmail"));
        return;
      }
      setStep("password");
      return;
    }

    setLoading(true);
    try {
      const supabase = createClient();
      const { error: err } = await supabase.auth.signInWithPassword({ email: email.trim(), password });
      if (err) throw err;
      window.location.href = redirectTo;
    } catch (err) {
      const msg = err instanceof Error ? err.message : t("common.genericError");
      setError(msg.toLowerCase().includes("invalid login credentials") ? t("login.badCredentials") : msg);
      setLoading(false);
    }
  }

  function change() {
    setStep("email");
    setPassword("");
    setShow(false);
    setError(null);
  }

  const locked = step === "password";

  return (
    <div>
      <h1 className="bw-auth__h1">{t("login.title")}</h1>
      <p className="mt-3 text-sm">
        <span className="text-muted-foreground">{t("login.newHere")} </span>
        <Link href="/signup" className="font-semibold text-primary underline-offset-4 hover:underline">{t("login.signUp")}</Link>
      </p>

      <div className="mt-8">
        <SocialButtons onError={setError} />
      </div>

      {error ? (
        <p role="alert" className="mt-5 rounded-sm border border-block/30 bg-block-muted px-3 py-2 text-xs text-block">
          {error}
        </p>
      ) : null}

      <form className="mt-5 space-y-4" onSubmit={onSubmit} noValidate>
        <div className="space-y-1.5">
          <Label htmlFor="email">{t("common.email")}</Label>
          <div className="relative">
            <Input
              ref={mailRef}
              id="email"
              name="email"
              type="email"
              autoComplete="username"
              value={email}
              onChange={(e) => setEmail(e.target.value)}
              readOnly={locked}
              tabIndex={locked ? -1 : 0}
              className={locked ? "bg-surface-3 pr-24 text-muted-foreground" : ""}
            />
            {locked ? (
              <button
                type="button"
                onClick={change}
                className="absolute right-3 top-1/2 -translate-y-1/2 text-xs font-semibold text-primary hover:underline"
              >
                {t("login.change")}
              </button>
            ) : null}
          </div>
        </div>

        {locked ? (
          <div className="space-y-1.5">
            <Label htmlFor="password">{t("common.password")}</Label>
            <div className="relative">
              <Input
                ref={pwRef}
                id="password"
                name="password"
                type={show ? "text" : "password"}
                autoComplete="current-password"
                value={password}
                onChange={(e) => setPassword(e.target.value)}
                className="pr-10"
              />
              <button
                type="button"
                onClick={() => setShow((v) => !v)}
                aria-label={show ? t("common.hide") : t("common.show")}
                className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
              >
                {show ? <EyeOff className="size-4" aria-hidden /> : <Eye className="size-4" aria-hidden />}
              </button>
            </div>
            <div className="flex justify-end pt-1">
              <Link href="/forgot-password" className="text-xs font-semibold text-primary underline-offset-4 hover:underline">
                {t("login.forgot")}
              </Link>
            </div>
          </div>
        ) : null}

        <button
          type="submit"
          disabled={loading || (locked ? password === "" : email.trim() === "")}
          className={`${btnPrimary} w-full disabled:opacity-60`}
        >
          {loading ? t("common.wait") : locked ? t("login.signIn") : t("login.next")}
        </button>
      </form>
    </div>
  );
}

"use client";

import { useEffect, useState } from "react";
import { useTranslations } from "next-intl";
import { Link } from "@/i18n/navigation";
import { WORKSPACE_STORAGE_KEY } from "@/components/aegisora/workspace-provider";
import { createClient } from "@/lib/supabase/client";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { btnOutline, btnPrimary } from "@/lib/ui";
import { cn } from "@/lib/utils";
import { Logo } from "@/components/aegisora/logo";
import { OnboardingIntro, INTRO_KEY } from "@/components/aegisora/onboarding-intro";

const STEP_KEYS = ["s0", "s1", "s2"];
type PolicyDecision = "ALLOW" | "BLOCK" | "ESCALATE";

export function OnboardingFlow() {
  const t = useTranslations("Onboarding");
  const steps = STEP_KEYS.map((k) => t("steps." + k));
  const [step, setStep] = useState(0);
  const [introDone, setIntroDone] = useState<boolean | null>(null);

  useEffect(() => {
    const timer = window.setTimeout(() => {
      try { setIntroDone(window.localStorage.getItem(INTRO_KEY) === "done"); } catch { setIntroDone(false); }
    }, 0);
    return () => window.clearTimeout(timer);
  }, []);
  const [workspace, setWorkspace] = useState("My workspace");
  const [workspaceId, setWorkspaceId] = useState<string | null>(null);
  const [agentName, setAgentName] = useState("finance-operator");
  const [actionPattern, setActionPattern] = useState("invoice.read");
  const [resourcePattern, setResourcePattern] = useState("invoice:sample-001");
  const [decision, setDecision] = useState<PolicyDecision>("ESCALATE");
  const [agentId, setAgentId] = useState<string | null>(null);
  const [policyId, setPolicyId] = useState<string | null>(null);
  const [aal, setAal] = useState<string | null>(null);
  const [aalReady, setAalReady] = useState(false);
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    if (step !== 1) return;
    let cancelled = false;
    void createClient().auth.mfa.getAuthenticatorAssuranceLevel().then(({ data, error: mfaError }) => {
      if (cancelled) return;
      setAal(mfaError ? null : data.currentLevel);
      setAalReady(true);
      if (mfaError) setError(t("err.mfa"));
    });
    return () => { cancelled = true; };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [step]);

  async function checkMfa() {
    setBusy(true);
    setError(null);
    const { data, error: mfaError } = await createClient().auth.mfa.getAuthenticatorAssuranceLevel();
    setBusy(false);
    setAalReady(true);
    if (mfaError) { setError(t("err.mfa")); return; }
    setAal(data.currentLevel);
  }

  async function createWorkspace() {
    setBusy(true);
    setError(null);
    const { data, error: createError } = await createClient().rpc("create_workspace", {
      p_name: workspace.trim(),
    });
    setBusy(false);
    if (createError || typeof data !== "string") {
      setError(createError?.message ?? t("err.noId"));
      return;
    }
    setWorkspaceId(data);
    window.localStorage.setItem(WORKSPACE_STORAGE_KEY, data);
    setStep(1);
  }

  async function postWorkspaceMutation(path: string, body: Record<string, unknown>) {
    if (!workspaceId) { setError(t("err.needWs")); return null; }
    setBusy(true);
    setError(null);
    try {
      const { data: { session } } = await createClient().auth.getSession();
      if (!session?.access_token) throw new Error(t("err.signIn"));
      const response = await fetch(path, {
        method: "POST",
        headers: { Authorization: `Bearer ${session.access_token}`, "Content-Type": "application/json" },
        body: JSON.stringify({ workspace_id: workspaceId, ...body }),
        cache: "no-store",
      });
      const result = await response.json().catch(() => null);
      if (!response.ok) {
        const message = result?.error?.message ?? t("err.failed");
        if (result?.error?.code === "mfa_required") setAal("aal1");
        throw new Error(message);
      }
      return result as Record<string, unknown>;
    } catch (cause) {
      setError(cause instanceof Error ? cause.message : t("err.failed"));
      return null;
    } finally {
      setBusy(false);
    }
  }

  async function createAgentAndRule() {
    let id = agentId;
    if (!id) {
      const agent = await postWorkspaceMutation("/api/v1/workspace/agents", {
        operation: "create",
        agent: { name: agentName.trim(), framework: null, description: "Created during workspace onboarding", external_id: null },
      });
      if (typeof agent?.id !== "string") return;
      id = agent.id;
      setAgentId(id);
    }
    const policy = await postWorkspaceMutation("/api/v1/workspace/policies", {
      name: `${agentName.trim()} onboarding rule`,
      agent_id: id,
      action_pattern: actionPattern.trim(),
      resource_pattern: resourcePattern.trim(),
      decision,
      priority: 100,
    });
    if (typeof policy?.id === "string") {
      setPolicyId(policy.id);
      setStep(2);
    }
  }

  if (introDone === null) return null;
  if (!introDone) return <OnboardingIntro onDone={() => setIntroDone(true)} />;

  return (
    <div className="mx-auto max-w-2xl px-6 py-12"><Logo className="mb-10" />
      <ol className="mb-10 grid grid-cols-3 gap-1">
        {steps.map((label, index) => (
          <li key={label} aria-current={index === step ? "step" : undefined}>
            <div className={cn("h-0.5", index <= step ? "bg-primary" : "bg-surface-4")} />
            <div className={cn("mt-2 font-mono text-[10px] tracking-wider", index === step ? "text-foreground" : "text-muted-foreground")}>
              {String(index + 1).padStart(2, "0")}<span className="hidden sm:inline"> {label.toUpperCase()}</span>
            </div>
          </li>
        ))}
      </ol>

      <section aria-labelledby="step-title" className="py-2">
        {step === 0 && (
          <>
            <h1 id="step-title" className="text-3xl font-semibold tracking-tight">{t("s0.title")}</h1>
            <p className="mt-2 text-sm text-muted-foreground">{t("s0.desc")}</p>
            <div className="mt-6 space-y-1.5"><Label htmlFor="workspace">{t("s0.label")}</Label><Input id="workspace" value={workspace} maxLength={120} onChange={(event) => setWorkspace(event.target.value)} /></div>
            <button type="button" onClick={() => void createWorkspace()} disabled={busy || !workspace.trim()} className={`${btnPrimary} mt-5 disabled:opacity-50`}>{busy ? t("s0.busy") : t("s0.btn")}</button>
          </>
        )}

        {step === 1 && (
          <>
            <h1 id="step-title" className="text-3xl font-semibold tracking-tight">{t("q.title")}</h1>
            <p className="mt-2 text-sm text-muted-foreground">{t("q.desc")}</p>

            {!aalReady ? (
              <p role="status" className="mt-6 text-sm text-muted-foreground">{t("q.checking")}</p>
            ) : aal !== "aal2" ? (
              <div className="mt-6 rounded-xl border border-border bg-card p-4">
                <p className="text-sm font-medium">{t("q.mfaTitle")}</p>
                <p className="mt-1 text-sm text-muted-foreground">{t("q.mfaDesc")}</p>
                <div className="mt-4 flex flex-wrap gap-3">
                  <Link href="/dashboard/identity" target="_blank" rel="noreferrer" className={btnPrimary}>{t("q.mfaOpen")}</Link>
                  <button type="button" onClick={() => void checkMfa()} disabled={busy} className={`${btnOutline} disabled:opacity-50`}>{busy ? t("q.checking") : t("q.mfaCheck")}</button>
                </div>

            ) : (

                  <div className="space-y-1.5"><Label htmlFor="action-pattern">{t("q.action")}</Label><Input id="action-pattern" value={actionPattern} maxLength={200} onChange={(event) => setActionPattern(event.target.value)} /></div>

                <fieldset className="mt-5 space-y-2"><legend className="mb-2 text-sm font-medium">{t("q.outcome")}</legend>
                  {(["ESCALATE", "BLOCK", "ALLOW"] as const).map((effect) => (
                    <label key={effect} className="flex cursor-pointer items-center gap-3 rounded-sm border border-border p-3 text-sm">
                      <input type="radio" name="decision" value={effect} checked={decision === effect} onChange={() => setDecision(effect)} />
                      <span><span className="font-mono text-xs font-semibold">{effect}</span> <span className="text-muted-foreground">{effect === "ESCALATE" ? t("q.escalate") : effect === "BLOCK" ? t("q.block") : t("q.allow")}</span></span>

                <button type="button" onClick={() => void createAgentAndRule()} disabled={busy || !agentName.trim() || !actionPattern.trim() || !resourcePattern.trim()} className={`${btnPrimary} mt-5 disabled:opacity-50`}>{busy ? t("q.busy") : t("q.create")}</button>
              </>

              {agentId && <div className="py-3"><dt className="text-xs text-muted-foreground">{t("finish.agent")}</dt><dd className="mt-1 text-sm">{agentName.trim()}</dd></div>}

type Answers = { companySize: string | null; role: string | null; stack: string[] };

async function persist(a: Answers) {
  try {
    window.localStorage.setItem(INTRO_KEY, "done");
    window.localStorage.setItem(INTRO_KEY + ":answers", JSON.stringify(a));
  } catch {}
  try {
    const supabase = createClient();
    const { data } = await supabase.auth.getUser();
    if (data.user) await supabase.from("profiles").update({ onboarding: a }).eq("id", data.user.id);
  } catch {}
}

function Chip({ label, active, onClick }: { label: string; active: boolean; onClick: () => void }) {

    >
      {label}
    </button>
  );
}

export function OnboardingIntro({ onDone }: { onDone: () => void }) {
  const t = useTranslations("OnboardingIntro");
  const total = 3;
  const [step, setStep] = useState(0);
  const [a, setA] = useState<Answers>({ companySize: null, role: null, stack: [] });
  const [q, setQ] = useState("");
  const [saving, setSaving] = useState(false);
  const timer = useRef<number | null>(null);

  const label = (s: Tile) => (s.nameKey ? t(s.nameKey) : (s.name ?? s.id));

  function pick(key: "companySize" | "role", value: string) {
    setA((cur) => ({ ...cur, [key]: value }));
    if (timer.current) window.clearTimeout(timer.current);
    timer.current = window.setTimeout(() => setStep((s) => Math.min(s + 1, total - 1)), 220);
  }
  function toggle(id: string) {
    setA((cur) => ({ ...cur, stack: cur.stack.includes(id) ? cur.stack.filter((x) => x !== id) : [...cur.stack, id] }));
  }
  async function finish() {
    setSaving(true);
    await persist(a);
    onDone();
  }

  const tiles = STACK.filter((s) => label(s).toLowerCase().includes(q.trim().toLowerCase()));

  return (
    <div className="min-h-screen bg-background">
      <header className="px-6 py-5 sm:px-10"><Logo /></header>
      <main className="mx-auto w-full max-w-3xl px-6 pb-20 pt-12 sm:pt-20">
        <div
          role="progressbar"
          aria-valuemin={1}
          aria-valuemax={total}
          aria-valuenow={step + 1}
          className="h-[3px] w-72 max-w-full overflow-hidden rounded-full bg-surface-4"
        >
          <div className="h-full rounded-full bg-primary transition-all duration-300" style={{ width: `${((step + 1) / total) * 100}%` }} />
        </div>

        {step === 0 && (
          <section aria-labelledby="intro-title">
            <h1 id="intro-title" className="mt-10 text-3xl font-semibold tracking-tight sm:text-4xl">{t("sizeTitle")}</h1>
            <p className="mt-3 max-w-xl text-sm text-muted-foreground">{t("sizeDesc")}</p>
            <div className="mt-8 flex flex-wrap gap-3">
              {SIZES.map((s) => <Chip key={s} label={s} active={a.companySize === s} onClick={() => pick("companySize", s)} />)}
            </div>
          </section>
        )}

        {step === 1 && (
          <section aria-labelledby="intro-title">
            <h1 id="intro-title" className="mt-10 text-3xl font-semibold tracking-tight sm:text-4xl">{t("roleTitle")}</h1>
            <p className="mt-3 max-w-xl text-sm text-muted-foreground">{t("roleDesc")}</p>
            <div className="mt-8 flex flex-wrap gap-3">
              {ROLES.map((r, i) => <Chip key={r} label={t("roles.r" + i)} active={a.role === r} onClick={() => pick("role", r)} />)}
            </div>
          </section>
        )}

        {step === 2 && (
          <section aria-labelledby="intro-title">
            <h1 id="intro-title" className="mt-10 text-3xl font-semibold tracking-tight sm:text-4xl">{t("stackTitle")}</h1>
            <p className="mt-3 max-w-xl text-sm text-muted-foreground">{t("stackDesc")}</p>
            <div className="mt-6 flex h-11 max-w-lg items-center gap-2 rounded-md border border-border bg-card px-3 focus-within:border-primary">
              <Search className="size-4 text-muted-foreground" aria-hidden="true" />
              <input
                value={q}
                onChange={(e) => setQ(e.target.value)}
                placeholder={t("search")}
                aria-label={t("search")}
                className="w-full bg-transparent text-sm outline-none placeholder:text-muted-foreground"
              />
            </div>
            <div className="mt-4 max-h-[26rem] overflow-y-auto rounded-xl border border-border p-4">
              {tiles.length === 0 ? (
                <p className="py-6 text-center text-sm text-muted-foreground">{t("noMatch")}</p>
              ) : (
                <ul className="grid grid-cols-2 gap-3 sm:grid-cols-3 md:grid-cols-4">
                  {tiles.map((tile) => {
                    const on = a.stack.includes(tile.id);
                    return (
                      <li key={tile.id}>
                        <button
                          type="button"
                          aria-pressed={on}
                          onClick={() => toggle(tile.id)}
                          className={cn(
                            "relative flex w-full flex-col items-center gap-2 rounded-lg border p-4 text-center text-xs font-medium transition",
                            focusRing,
                            on ? "border-primary bg-primary/5" : "border-border bg-card hover:border-foreground/30",
                          )}
                        >
                          <span className="grid size-12 place-items-center rounded-lg bg-primary/10 font-mono text-sm font-semibold text-primary">{tile.mark}</span>
                          {label(tile)}
                          {on && (
                            <span className="absolute right-2 top-2 grid size-4 place-items-center rounded-full bg-primary text-primary-foreground">
                              <Check className="size-3" aria-hidden="true" />
                            </span>
                          )}
                        </button>
                      </li>
                    );
                  })}
                </ul>
              )}
            </div>
            <div className="mt-6">
              <button type="button" onClick={() => void finish()} disabled={saving} className={`${btnPrimary} gap-2 disabled:opacity-50`}>
                {saving ? t("saving") : a.stack.length ? t("contN", { n: a.stack.length }) : t("cont")}
                <ArrowRight className="size-4" aria-hidden="true" />
              </button>
            </div>
          </section>
        )}

        <div className="mt-10 flex items-center gap-6 text-sm text-muted-foreground">
          {step > 0 && (
            <button type="button" onClick={() => setStep((s) => Math.max(0, s - 1))} className={cn("underline-offset-4 hover:text-foreground hover:underline", focusRing)}>{t("back")}</button>
          )}
          <button type="button" onClick={() => void finish()} className={cn("underline-offset-4 hover:text-foreground hover:underline", focusRing)}>{t("skip")}</button>
        </div>
      </main>
    </div>
  );
}

export function SignupForm({ redirectTo }: { redirectTo: string }) {
  const t = useTranslations("Auth");
  const [email, setEmail] = useState("");
  const [first, setFirst] = useState("");
  const [last, setLast] = useState("");
  const [password, setPassword] = useState("");
  const [company, setCompany] = useState("");
  const [show, setShow] = useState(false);
  const [accepted, setAccepted] = useState(false);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [notice, setNotice] = useState<string | null>(null);

  const mail = email.trim().toLowerCase();
  const longEnough = password.length >= 8;
  const noEmail = password.length > 0 && !(mail && password.toLowerCase().includes(mail));
  const valid = Boolean(EMAIL_RE.test(email.trim()) && first.trim() && last.trim() && longEnough && noEmail && accepted);

  async function onSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    if (!valid) return;
    setError(null);
    setNotice(null);
    setLoading(true);
    try {
      const firstName = first.trim();
      const lastName = last.trim();
      const supabase = createClient();
      // handle_new_user tetikleyicisi company bilgisinden workspace olusturur.
      // Bos birakilirsa tetikleyicinin bos deger gormemesi icin varsayilan bir ad gonderilir.
      const { data, error: err } = await supabase.auth.signUp({
        email: email.trim(),
        password,
        options: {
          data: {
            full_name: firstName + " " + lastName,
            first_name: firstName,
            last_name: lastName,
            company: company.trim() || firstName + "'s workspace",
          },
        },
      });
      if (err) throw err;
      if (data.user && Array.isArray(data.user.identities) && data.user.identities.length === 0) {
        throw new Error(t("signup.exists"));
      }
      if (data.session) {
        window.location.href = redirectTo;
        return;
      }
      setNotice(t("signup.checkInbox"));
      setLoading(false);
    } catch (err) {
      const raw = err instanceof Error ? err.message : "";
      if (/database error saving new user/i.test(raw)) {
        const de = document.documentElement.lang === "de";
        setError(de
          ? "Aegisora ist derzeit nur für freigegebene Pilotnutzer geöffnet. Bitte bewirb dich über die Pilot-Seite."
          : "Aegisora is currently open to approved pilot users only. Please apply on the pilot page.");
      } else {
        setError(raw || t("common.genericError"));
      }
      setLoading(false);
    }
  }

  return (
    <div>
      <h1 className="bw-auth__h1">{t("signup.title")}</h1>
      <p className="mt-3 text-sm">
        <span className="text-muted-foreground">{t("signup.have")} </span>
        <Link href="/login" className="font-semibold text-primary underline-offset-4 hover:underline">{t("signup.logIn")}</Link>
      </p>

      <div className="mt-8">
        <SocialButtons onError={setError} />
      </div>

      {error ? (
        <p role="alert" className="mt-5 rounded-sm border border-block/30 bg-block-muted px-3 py-2 text-xs text-block">
          {error}
        </p>
      ) : null}
      {notice ? (
        <p role="status" className="mt-5 rounded-sm border border-cyan/30 bg-surface-3 px-3 py-2 text-xs text-cyan">
          {notice}
        </p>
      ) : null}

      <form className="mt-5 space-y-4" onSubmit={onSubmit} noValidate>
        <div className="space-y-1.5">
          <Label htmlFor="email">{t("common.email")}</Label>
          <p className="text-xs text-muted-foreground">{t("signup.emailHint")}</p>
          <Input id="email" name="email" type="email" autoComplete="email" value={email} onChange={(e) => setEmail(e.target.value)} />
        </div>

        <div className="grid gap-4 sm:grid-cols-2">
          <div className="space-y-1.5">
            <Label htmlFor="first">{t("signup.first")}</Label>
            <Input id="first" name="first" type="text" autoComplete="given-name" value={first} onChange={(e) => setFirst(e.target.value)} />
          </div>
          <div className="space-y-1.5">
            <Label htmlFor="last">{t("signup.last")}</Label>
            <Input id="last" name="last" type="text" autoComplete="family-name" value={last} onChange={(e) => setLast(e.target.value)} />
          </div>
        </div>

        <div className="space-y-1.5">
          <Label htmlFor="password">{t("common.password")}</Label>
          <div className="relative">
            <Input
              id="password"
              name="password"
              type={show ? "text" : "password"}
              autoComplete="new-password"
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              className="pr-10"
            />
            <button
              type="button"
              onClick={() => setShow((v) => !v)}
              aria-label={show ? t("common.hide") : t("common.show")}
              className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
            >
              {show ? <EyeOff className="size-4" aria-hidden /> : <Eye className="size-4" aria-hidden />}
            </button>
          </div>
          <ul className="space-y-1 pt-1">
            <Rule ok={longEnough}>{t("signup.ruleLen")}</Rule>
            <Rule ok={noEmail}>{t("signup.ruleEmail")}</Rule>
          </ul>
        </div>

        <div className="space-y-1.5">
          <Label htmlFor="company">{t("signup.companyLabel")} <span className="text-xs font-normal text-muted-foreground">{t("signup.optional")}</span></Label>
          <Input id="company" name="company" type="text" autoComplete="organization" value={company} onChange={(e) => setCompany(e.target.value)} />
        </div>

        <label className="flex items-start gap-3 text-sm leading-5">
          <input type="checkbox" checked={accepted} onChange={(e) => setAccepted(e.target.checked)} className="mt-0.5 size-4 accent-primary" />
          <span>
            {t.rich("signup.accept", {
              privacy: (chunks) => (
                <Link href="/trust/privacy" target="_blank" className="text-primary underline-offset-4 hover:underline">{chunks}</Link>
              ),
              terms: (chunks) => (
                <Link href="/terms" target="_blank" className="text-primary underline-offset-4 hover:underline">{chunks}</Link>
              ),
            })}
          </span>
        </label>

        <button type="submit" disabled={loading || !valid} className={`${btnPrimary} w-full disabled:opacity-60`}>
          {loading ? t("common.wait") : t("signup.create")}
        </button>
      </form>
    </div>
  );
}

"use client";

import { useEffect, useState } from "react";
import { Volume2, VolumeX } from "lucide-react";
import { play, setSoundEnabled, soundEnabled, type SoundKind } from "@/lib/aegisora/sound";

const TARGET = 'button, a[href], [role="button"], [role="switch"], [role="tab"], [role="menuitem"], input[type="checkbox"], input[type="radio"], summary';
const DANGER = /delete|remove|revoke|block|disable|cancel|reject|deny|sil\b|kald/i;
const CONFIRM = /save|create|add|approve|confirm|send|invite|connect|import|apply|generate|export/i;

function pick(el: HTMLElement): SoundKind | null {
  const mode = el.getAttribute("data-sound");
  if (mode === "off") return null;
  if (mode === "confirm" || mode === "danger" || mode === "nav" || mode === "click") return mode;
  if (el.matches('[role="switch"], input[type="checkbox"]')) {
    const on = el.getAttribute("aria-checked") === "true" || (el as HTMLInputElement).checked;
    return on ? "off" : "on";
  }
  const label = (el.getAttribute("aria-label") || el.textContent || "").trim().slice(0, 60);
  if (DANGER.test(label)) return "danger";
  if (el.matches("a[href]")) return "nav";
  if ((el as HTMLButtonElement).type === "submit" || CONFIRM.test(label)) return "confirm";
  return "click";
}

export function SoundProvider() {
  useEffect(() => {
    const onClick = (e: MouseEvent) => {
      const t = e.target;
      if (!(t instanceof Element)) return;
      const el = t.closest<HTMLElement>(TARGET);
      if (!el) return;
      if ((el as HTMLButtonElement).disabled || el.getAttribute("aria-disabled") === "true") return;
      const kind = pick(el);
      if (kind) play(kind);
    };
    document.addEventListener("click", onClick, true);
    return () => document.removeEventListener("click", onClick, true);
  }, []);
  return null;
}

export function SoundToggle() {
  const [on, setOn] = useState(true);
  useEffect(() => {
    const id = window.setTimeout(() => setOn(soundEnabled()), 0);
    return () => window.clearTimeout(id);
  }, []);
  const Icon = on ? Volume2 : VolumeX;
  return (
    <button
      type="button"
      data-sound="off"
      aria-label={on ? "Mute interface sounds" : "Unmute interface sounds"}
      aria-pressed={!on}
      title={on ? "Sounds on" : "Sounds off"}
      onClick={() => { const next = !on; setSoundEnabled(next); setOn(next); if (next) play("on"); }}

/** Replaces window.confirm. Resolves true only when the person confirms. */
export function confirmDialog(options: ConfirmOptions): Promise<boolean> {
  return new Promise((resolve) => {
    if (!listener) {
      resolve(false);
      return;
    }
    listener.onConfirm({ ...options, resolve });
  });
}

/** Replaces window.alert. Shows a short message in the corner. */
export function notify(text: string, ok = true) {
  listener?.onToast({ id: ++seq, text, ok });
}

export function FeedbackHost() {
  const [req, setReq] = useState<ConfirmRequest | null>(null);
  const [toasts, setToasts] = useState<Toast[]>([]);
  const active = useRef<ConfirmRequest | null>(null);
  const pending = useRef<ConfirmRequest[]>([]);
  const cancelRef = useRef<HTMLButtonElement>(null);

  useEffect(() => {
    listener = {
      onConfirm: (request) => {
        if (active.current) {
          pending.current.push(request);
        } else {
          active.current = request;
          setReq(request);
        }
      },
      onToast: (toast) => {
        setToasts((list) => [...list.slice(-2), toast]);
        window.setTimeout(() => setToasts((list) => list.filter((x) => x.id !== toast.id)), toast.ok ? 4000 : 7000);
      },
    };
    return () => {
      listener = null;
    };

    if (!current) return;
    current.resolve(value);
    const next = pending.current.shift() ?? null;
    active.current = next;

export type SoundKind = "click" | "nav" | "confirm" | "danger" | "on" | "off";

const KEY = "aeg-sound";
const MASTER = 0.5;
let ctx: AudioContext | null = null;
let last = 0;

export function soundEnabled(): boolean {
  try { return localStorage.getItem(KEY) !== "off"; } catch { return true; }
}

export function setSoundEnabled(on: boolean) {
  try { localStorage.setItem(KEY, on ? "on" : "off"); } catch { /* storage unavailable */ }
}

function tone(c: AudioContext, freq: number, start: number, dur: number, vol: number, endFreq?: number, type: OscillatorType = "sine") {
  const o = c.createOscillator();
  const g = c.createGain();
  o.type = type;
  o.frequency.setValueAtTime(freq, start);
  if (endFreq) o.frequency.exponentialRampToValueAtTime(endFreq, start + dur);
  g.gain.setValueAtTime(0.0001, start);
  g.gain.exponentialRampToValueAtTime(vol * MASTER, start + 0.008);
  g.gain.exponentialRampToValueAtTime(0.0001, start + dur);
  o.connect(g);
  g.connect(c.destination);
  o.start(start);
  o.stop(start + dur + 0.02);
}

export function play(kind: SoundKind) {
  if (typeof window === "undefined" || !soundEnabled()) return;
  const now = performance.now();
  if (now - last < 45) return;
  last = now;
  try {
    const AC = window.AudioContext || (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
    if (!AC) return;
    ctx = ctx ?? new AC();
    if (ctx.state === "suspended") void ctx.resume();
    const t = ctx.currentTime;
    switch (kind) {
      case "click": tone(ctx, 720, t, 0.07, 0.18, 560); break;
      case "nav": tone(ctx, 520, t, 0.09, 0.16, 780); break;
      case "confirm": tone(ctx, 660, t, 0.1, 0.2); tone(ctx, 990, t + 0.07, 0.16, 0.2); break;
      case "danger": tone(ctx, 300, t, 0.16, 0.2, 200, "triangle"); break;
      case "on": tone(ctx, 600, t, 0.07, 0.18); tone(ctx, 900, t + 0.05, 0.09, 0.18); break;
      case "off": tone(ctx, 900, t, 0.07, 0.18); tone(ctx, 600, t + 0.05, 0.09, 0.18); break;
    }
  } catch { /* audio blocked: stay silent */ }
}

export const dynamic = "force-dynamic";

function errorResponse(status: number, code: string, message: string) {
  return NextResponse.json({ error: { code, message } }, { status });
}

export async function POST(req: Request) {
  try {
    const { user, aal } = await authenticateUser(req);
    const body = await readJson(req, 512_000);
    const workspaceId = typeof body.workspace_id === "string" ? body.workspace_id : "";
    if (!/^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i.test(workspaceId)) {
      throw new ApiError(400, "invalid_request", "'workspace_id' must be a UUID.");
    }
    await requireWorkspacePermission(user.id, workspaceId, "governance:manage");
    requireAal2(aal);

    const operation = body.operation;
    const db = admin();
    if (operation === "create") {
      const normalized = normalizeAgentInput(body.agent, workspaceId, "manual");
      if (!normalized.row) throw new ApiError(400, "invalid_agent", normalized.error ?? "Invalid agent.");
      await assertWithinPlanLimit(workspaceId, "agents", 1);
      const { data, error } = await db.from("agents").insert(normalized.row).select("id").single();
      if (error) throw error;
      return NextResponse.json({ id: data.id }, { status: 201 });
    }

    if (operation === "import") {
      const agents = Array.isArray(body.agents) ? body.agents : null;
      if (!agents || agents.length === 0 || agents.length > 500) {
        throw new ApiError(400, "invalid_request", "'agents' must be an array of 1–500 items.");
      }
      const rows = new Map<string, AgentWrite>();
      for (const agent of agents) {
        const normalized = normalizeAgentInput(agent, workspaceId, "import");
        if (!normalized.row) throw new ApiError(400, "invalid_agent", normalized.error ?? "Invalid agent in import.");
        rows.set(normalized.row.name, normalized.row);
      }
      await assertAgentImportWithinLimit(workspaceId, [...rows.keys()]);
      const { error } = await db.from("agents").upsert([...rows.values()], { onConflict: "workspace_id,name" });
      if (error) throw error;
      return NextResponse.json({ imported: rows.size });
    }

    if (operation === "update") {
      const agentId = typeof body.agent_id === "string" ? body.agent_id : "";
      const state = body.state;
      if (!/^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i.test(agentId) || (state !== "ACTIVE" && state !== "DISABLED")) {
        throw new ApiError(400, "invalid_request", "'agent_id' and a valid state are required.");
      }
      const { data, error } = await db.from("agents").update({ state }).eq("id", agentId).eq("workspace_id", workspaceId).select("id").maybeSingle();
      if (error) throw error;
      if (!data) throw new ApiError(404, "not_found", "Agent not found in this workspace.");
      return NextResponse.json({ updated: true });
    }

    if (operation === "delete") {
      const agentId = typeof body.agent_id === "string" ? body.agent_id : "";
      if (!/^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i.test(agentId)) {
        throw new ApiError(400, "invalid_request", "'agent_id' must be a UUID.");
      }
      const { data, error } = await db.from("agents").delete().eq("id", agentId).eq("workspace_id", workspaceId).select("id").maybeSingle();
      if (error) throw error;
      if (!data) throw new ApiError(404, "not_found", "Agent not found in this workspace.");
      return NextResponse.json({ deleted: true });
    }

    throw new ApiError(400, "invalid_request", "Unsupported agent operation.");
  } catch (error) {

    }
    if (error instanceof ApiError) return errorResponse(error.status, error.code, error.message);
    console.error("[aegisora] Workspace agent operation failed.", error);
    return errorResponse(500, "internal_error", "Unable to complete the workspace agent operation.");
