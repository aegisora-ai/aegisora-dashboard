  if (recent.length >= MAX_PER_WINDOW) {
    return Response.json({ error: "Too many requests. Please try again later." }, { status: 429 });
  }
  recent.push(now);

"use client";

import { useState } from "react";
import { useTranslations } from "next-intl";
import { Link, useRouter } from "@/i18n/navigation";
import { createClient } from "@/lib/supabase/client";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { btnPrimary } from "@/lib/ui";

export default function ResetPasswordPage() {
  const t = useTranslations("Auth");
  const router = useRouter();
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [done, setDone] = useState(false);

  async function onSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    setError(null);
    setLoading(true);
    const form = new FormData(e.currentTarget);
    const password = String(form.get("password") ?? "");

    const supabase = createClient();
    const { error: err } = await supabase.auth.updateUser({ password });
    setLoading(false);
    if (err) {
      setError(err.message);
      return;
    }
    setDone(true);
    setTimeout(() => router.push("/dashboard"), 1500);
  }

  return (
    <div className="w-full max-w-sm">
      <h1 className="bw-auth__h1">{t("reset.title")}</h1>
      <p className="mt-3 text-sm text-muted-foreground">{t("reset.desc")}</p>

      {error && (
        <p role="alert" className="mt-6 rounded-sm border border-block/30 bg-block-muted px-3 py-2 text-xs text-block">
          {error}
        </p>
      )}
      {done && (
        <p role="status" className="mt-6 rounded-sm border border-allow/30 bg-allow-muted px-3 py-2 text-xs text-allow">
          {t("reset.done")}
        </p>
      )}

      <form className="mt-6 space-y-4" onSubmit={onSubmit}>
        <div className="space-y-1.5">
          <Label htmlFor="password">{t("reset.label")}</Label>
          <Input id="password" name="password" type="password" autoComplete="new-password" required minLength={8} />
        </div>
        <button type="submit" disabled={loading || done} className={`${btnPrimary} w-full disabled:opacity-60`}>
          {loading ? t("reset.updating") : t("reset.submit")}
        </button>
      </form>

      <p className="mt-6 text-sm text-muted-foreground">
        <Link href="/login" className="text-primary underline-offset-4 hover:underline">{t("common.backToLogin")}</Link>

function NavGroupBlock({ group, defaultOpen }: { group: NavGroup; defaultOpen: boolean }) {
  const pathname = usePathname();
  const t = useTranslations('Docs');
  const [open, setOpen] = useState(defaultOpen);

  return (
    <div className="border-b border-border/60 pb-1 last:border-0">
      <button
        type="button"
        aria-expanded={open}
        onClick={() => setOpen((o) => !o)}
        className="flex w-full items-center justify-between py-2.5 text-left transition-colors hover:text-foreground"
      >
        <span className="font-mono text-[11px] font-semibold uppercase tracking-wider text-muted-foreground">
          {group.key ? t('groups.' + group.key) : group.title}
        </span>
        <ChevronRight
          size={14}
          aria-hidden
          className={cn(
            'shrink-0 text-muted-foreground transition-transform duration-300 ease-out',
            open && 'rotate-90'
          )}
        />
      </button>

      <div
        className={cn(
          'grid transition-[grid-template-rows] duration-300 ease-in-out',
          open ? 'grid-rows-[1fr]' : 'grid-rows-[0fr]'
        )}
      >
        <div className="overflow-hidden">
          <ul className="flex flex-col gap-0.5 pb-3 pl-1">
            {group.items.map((item) => {
              // Only the page itself counts as active; #anchor links never do.
              const active = !item.href.includes('#') && pathname === item.href;
              return (
                <li key={item.title}>
                  <Link
                    href={item.href}
                    aria-current={active ? 'page' : undefined}
                    className={cn(
                      'block rounded-md px-3 py-1.5 text-sm transition-colors',
                      active
                        ? 'bg-accent font-medium text-accent-foreground'
                        : 'text-muted-foreground hover:bg-accent/60 hover:text-foreground'
                    )}
                  >
                    {item.key ? t('items.' + item.key) : item.title}
                  </Link>
                </li>
              );
            })}
          </ul>
        </div>
      </div>
    </div>
  );
}

export function DocsNav() {
  const t = useTranslations('Docs');
  return (
    <nav aria-label={t('aria')} className="max-h-[calc(100vh-7rem)] overflow-y-auto pr-2">
      {NAV_TREE.map((group, i) => (
        <NavGroupBlock key={group.title} group={group} defaultOpen={i < 2} />
      ))}
    </nav>
  );
}

function sanitize(raw: unknown): Prefs {
  const r = (raw ?? {}) as Partial<Prefs>;
  const tabs = Array.isArray(r.tabs) ? r.tabs.filter((t) => typeof t === "string" && byId(t)) : [];
  return {
    tabs: tabs.length === 4 ? tabs : DEFAULTS.tabs,
    density: r.density === "compact" || r.density === "large" ? r.density : "standard",
    home: typeof r.home === "string" && byId(r.home) ? r.home : "overview",
  };
}
