"use client";

import Link from "next/link";
import { useEffect, useState } from "react";
import { CircleHelp, Moon, Sun, Volume2, VolumeX } from "lucide-react";
import { THEME_KEY } from "@/lib/theme";
import { play, setSoundEnabled, soundEnabled } from "@/lib/aegisora/sound";
import { focusRing } from "@/lib/ui";
import { cn } from "@/lib/utils";

const btn = "inline-flex size-9 items-center justify-center rounded-sm border border-border text-muted-foreground hover:bg-surface-3 hover:text-foreground";

export function TopbarActions() {
  const [dark, setDark] = useState(true);
  const [snd, setSnd] = useState(true);

  useEffect(() => {
    const t = window.setTimeout(() => { setDark(document.documentElement.classList.contains("dark")); setSnd(soundEnabled()); }, 0);
    return () => window.clearTimeout(t);
  }, []);

  function toggle() {
    const next = !dark;
    setDark(next);
    document.documentElement.classList.toggle("dark", next);
    try { window.localStorage.setItem(THEME_KEY, next ? "dark" : "light"); } catch {}
  }

  return (
    <>
      <span className="hidden sm:inline-flex">
        <Link href="/support" aria-label="Help center" className={cn(btn, focusRing)}>
          <CircleHelp className="size-4" />
        </Link>
      </span>
      <button
        type="button"
        data-sound="off"
        onClick={() => { const next = !snd; setSoundEnabled(next); setSnd(next); if (next) play("on"); }}
        aria-label={snd ? "Mute interface sounds" : "Unmute interface sounds"}
        aria-pressed={!snd}
        className={cn(btn, focusRing)}
      >
        {snd ? <Volume2 className="size-4" /> : <VolumeX className="size-4" />}
      </button>
      <button
        type="button"
        onClick={toggle}
        aria-label={dark ? "Switch to light theme" : "Switch to dark theme"}
        className={cn(btn, focusRing)}
      >
        {dark ? <Sun className="size-4" /> : <Moon className="size-4" />}
      </button>
    </>
  );
}

"use client";

import { createContext, useCallback, useContext, useEffect, useState } from "react";
import { createClient } from "@/lib/supabase/client";

export type WorkspaceSummary = { id: string; name: string; slug: string; role: string };

type WorkspaceContextValue = {
  loading: boolean;
  userEmail: string | null;
  userName: string | null;
  workspaces: WorkspaceSummary[];
  currentWorkspace: WorkspaceSummary | null;
  setCurrentWorkspaceId: (id: string) => void;
  refresh: () => void;
};

const WorkspaceContext = createContext<WorkspaceContextValue | null>(null);

export const WORKSPACE_STORAGE_KEY = "aeg-current-workspace";

export function WorkspaceProvider({ children }: { children: React.ReactNode }) {
  const [loading, setLoading] = useState(true);
  const [userEmail, setUserEmail] = useState<string | null>(null);
  const [userName, setUserName] = useState<string | null>(null);
  const [workspaces, setWorkspaces] = useState<WorkspaceSummary[]>([]);
  const [currentId, setCurrentId] = useState<string | null>(null);

  const load = useCallback(async () => {
    setLoading(true);
    const supabase = createClient();

    const { data: userData } = await supabase.auth.getUser();
    const user = userData.user;
    setUserEmail(user?.email ?? null);
    setUserName((user?.user_metadata?.full_name as string) ?? null);

    if (!user) {
      setWorkspaces([]);

"use client";

import { useCallback, useEffect, useState } from "react";
import { createClient } from "@/lib/supabase/client";

export type Workspace = { id: string; name: string; slug: string };
let cached: Workspace | null = null;

export function useWorkspace() {
  const [ws, setWs] = useState<Workspace | null>(cached);
  const [loading, setLoading] = useState(cached === null);
  const [error, setError] = useState<string | null>(null);

  const load = useCallback(async () => {
    setLoading(true);
    const supabase = createClient();
    const { data, error: err } = await supabase
      .from("workspace_members")
      .select("workspaces(id,name,slug)")
      .order("created_at", { ascending: true })
      .limit(1);
    if (err) {
      setError(err.message);
      setLoading(false);
      return;
    }
    const raw = (data?.[0] as unknown as { workspaces?: Workspace | Workspace[] } | undefined)?.workspaces;
    cached = (Array.isArray(raw) ? raw[0] : raw) ?? null;
    setWs(cached);
    setError(null);
    setLoading(false);
  }, []);

  // Cache miss intentionally triggers asynchronous workspace hydration.
  useEffect(() => {
    if (cached === null) {
      const timer = window.setTimeout(() => {
        void load();
      }, 0);

      return () => window.clearTimeout(timer);
    }

    return undefined;
  }, [load]);

  return { ws, loading, error, reload: load };
}

export function usePendingApprovals(): number {
  const { ws } = useWorkspace();
  const [n, setN] = useState(0);

  useEffect(() => {
    if (!ws) return;
    const supabase = createClient();
    let alive = true;
    const load = async () => {
      const { count } = await supabase
        .from("approvals")
        .select("id", { count: "exact", head: true })
        .eq("workspace_id", ws.id)
        .eq("status", "pending")
        .gt("expires_at", new Date().toISOString());
      if (alive) setN(count ?? 0);
    };
    void load();
    const ch = supabase
      .channel(`pending-${ws.id}-${Math.random().toString(36).slice(2)}`)
      .on("postgres_changes", { event: "*", schema: "public", table: "approvals", filter: `workspace_id=eq.${ws.id}` }, () => { void load(); })
      .subscribe();
    return () => {
      alive = false;
      void supabase.removeChannel(ch);
    };
  }, [ws]);

  return n;
}

function parseImport(text: string): Parsed[] {
  const t = text.trim();
  if (!t) return [];
  if (t.startsWith("[") || t.startsWith("{")) {
    const v = JSON.parse(t);
    const list: Record<string, unknown>[] = Array.isArray(v) ? v : Array.isArray(v?.agents) ? v.agents : [];
    return list
      .filter((a) => a && typeof a.name === "string" && a.name.trim())
      .map((a) => ({
        name: String(a.name).trim().slice(0, 120),

        metadata: a.metadata && typeof a.metadata === "object" ? (a.metadata as object) : {},
      }));
  }
  return t
    .split(/\r?\n/)
    .map((l) => l.split(",").map((x) => x.trim()))
    .filter((c) => c[0] && c[0].toLowerCase() !== "name")

                    <Link href={item.href} className="text-sm text-slate-300 transition-colors hover:text-white">
                      {item.title}
                    </Link>
                  </li>
                ))}
              </ul>
            </div>
          ))}
        </div>
        <div className="mt-5 flex items-center justify-between gap-3 border-t border-slate-800 pt-4">
          <p className="text-xs text-slate-500">Full documentation, versioned and searchable.</p>
          <Link href="/docs" className="shrink-0 text-xs font-semibold text-[#1677ff] hover:underline">
            Open Docs &rarr;
          </Link>
        </div>
      </div>
    </div>
  );
}
